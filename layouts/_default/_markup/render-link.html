{{- /* Hugo Goldmark render hook: Markdown 链接 → 与 render-image.html 同逻辑：避免 `"/foo" | relURL` 在 render hook 内 不补 basePath 的坑，手工从 site.BaseURL 取 path 前缀拼接。 */ -}} {{- $basePath := (urls.Parse site.BaseURL).Path | default "/" -}} {{- $basePath = strings.TrimSuffix "/" $basePath -}} {{- $dest := .Destination -}} {{- $isExternal := or (strings.HasPrefix $dest "http://") (strings.HasPrefix $dest "https://") -}} {{- if not $isExternal -}} {{- if strings.HasPrefix $dest "/" -}} {{- if $basePath -}} {{- if not (strings.HasPrefix $dest (printf "%s/" $basePath)) -}} {{- $dest = printf "%s%s" $basePath $dest -}} {{- end -}} {{- end -}} {{- else -}} {{- $dest = $dest | relURL -}} {{- end -}} {{- end -}} {{- $text := .Text -}} {{- $title := .Title -}} {{- $text | safeHTML -}}
